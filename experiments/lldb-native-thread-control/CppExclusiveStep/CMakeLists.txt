cmake_minimum_required(VERSION 3.16)
project(CppExclusiveStep LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(cpp-exclusive-step main.cpp)
add_executable(cpp-blocked-step blocked.cpp)
target_compile_features(cpp-exclusive-step PRIVATE cxx_std_17)
target_compile_options(cpp-exclusive-step PRIVATE -g -O0)
target_link_libraries(cpp-exclusive-step PRIVATE Threads::Threads)
target_compile_features(cpp-blocked-step PRIVATE cxx_std_17)
target_compile_options(cpp-blocked-step PRIVATE -g -O0)
target_link_libraries(cpp-blocked-step PRIVATE Threads::Threads)
