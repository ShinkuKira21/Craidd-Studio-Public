cmake_minimum_required(VERSION 3.16)
project(MtCppGui LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
add_executable(mt_cpp_gui main.cpp)
target_compile_features(mt_cpp_gui PRIVATE cxx_std_17)
target_compile_options(mt_cpp_gui PRIVATE -g -O0)
target_link_libraries(mt_cpp_gui PRIVATE Threads::Threads PkgConfig::GTK4)
